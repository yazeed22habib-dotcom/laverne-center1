// ========================================== // LAVERNE CENTER - BACKEND V2 // نظام الحجوزات المستقلة لكل قسم // ========================================== var ADMIN_PIN = "1234"; var ARCHIVE_VERSION = "LAVERNE_BOOKINGS_RESET_V2"; function getDB() { var props = PropertiesService.getScriptProperties(); var ssId = props.getProperty("SS_ID"); var ss; if (ssId) { try { ss = SpreadsheetApp.openById(ssId); } catch (e) {} } if (!ss) { ss = SpreadsheetApp.create("حجوزات وإعدادات مركز لافيرن"); props.setProperty("SS_ID", ss.getId()); } var bookings = ss.getSheetByName("Bookings"); if (!bookings) { bookings = ss.insertSheet("Bookings"); bookings.appendRow([ "ID", "التاريخ", "الوقت", "الاسم", "الهاتف", "الخدمات", "الإجمالي", "الحالة", "الأقسام" ]); bookings.setFrozenRows(1); } var config = ss.getSheetByName("Config"); if (!config) { config = ss.insertSheet("Config"); config.appendRow(["Key", "Value"]); } // أرشفة الحجوزات القديمة مرة واحدة فقط archiveOldBookings(ss); return ss; } // ========================================== // أرشفة الحجوزات القديمة // ========================================== function archiveOldBookings(ss) { var props = PropertiesService.getScriptProperties(); if (props.getProperty(ARCHIVE_VERSION) === "DONE") { return; } var lock = LockService.getScriptLock(); lock.waitLock(20000); try { if (props.getProperty(ARCHIVE_VERSION) === "DONE") { return; } var sh = ss.getSheetByName("Bookings"); if (!sh) return; var last = sh.getLastRow(); if (last > 1) { var archive = ss.getSheetByName("Old Bookings"); if (!archive) { archive = ss.insertSheet("Old Bookings"); } var oldData = sh.getRange(1, 1, last, sh.getLastColumn()).getValues(); if (archive.getLastRow() === 0) { archive.getRange(1, 1, oldData.length, oldData[0].length) .setValues(oldData); } else { var dataOnly = oldData.slice(1); if (dataOnly.length) { archive.getRange( archive.getLastRow() + 1, 1, dataOnly.length, dataOnly[0].length ).setValues(dataOnly); } } // تفريغ الحجوزات الحالية مع الاحتفاظ بالعناوين if (last > 1) { sh.deleteRows(2, last - 1); } } sh.getRange(1, 1, 1, 9).setValues([[ "ID", "التاريخ", "الوقت", "الاسم", "الهاتف", "الخدمات", "الإجمالي", "الحالة", "الأقسام" ]]); sh.setFrozenRows(1); props.setProperty(ARCHIVE_VERSION, "DONE"); } finally { lock.releaseLock(); } } // ========================================== // استقبال طلبات الموقع // ========================================== function doGet(e) { try { var ss = getDB(); var cfg = getConfig(ss); var booked = getBookedMap(ss); return jsonOut({ ok: true, cfg: cfg, booked: booked }); } catch (err) { return jsonOut({ ok: false, message: err.toString() }); } } function doPost(e) { try { var data = JSON.parse(e.postData.contents || "{}"); var action = data.a; var ss = getDB(); // ====================================== // جلب الحجوزات // ====================================== if (action === "list") { if (!checkPin(data.pin)) { return jsonOut({ ok: false, message: "الرقم السري غير صحيح" }); } return jsonOut({ ok: true, rows: getRows(ss) }); } // ====================================== // حجز موعد جديد // ====================================== if (action === "book") { var lock = LockService.getScriptLock(); lock.waitLock(20000); try { var date = String(data.date || "").trim(); var time = String(data.time || "").trim(); var name = String(data.name || "").trim(); var phone = String(data.phone || "").trim(); if (!date || !time || !name || !phone) { return jsonOut({ ok: false, message: "يرجى استكمال بيانات الحجز" }); } var categoryKeys = data.categoryKeys || []; if (!Array.isArray(categoryKeys) || !categoryKeys.length) { categoryKeys = ["*"]; } categoryKeys = categoryKeys.map(function(k) { return String(k); }); var booked = getBookedMap(ss); var day = booked[date] || {}; // فحص جميع الأقسام المختارة for (var i = 0; i < categoryKeys.length; i++) { var key = categoryKeys[i]; if ( (day["*"] && day["*"].indexOf(time) !== -1) || (day[key] && day[key].indexOf(time) !== -1) ) { return jsonOut({ ok: false, err: "taken", message: "الموعد غير متاح في أحد الأقسام المختارة" }); } } var sh = ss.getSheetByName("Bookings"); sh.appendRow([ Utilities.getUuid(), date, time, name, "'" + phone, String(data.services || ""), String(data.totalText || data.total || ""), "active", JSON.stringify(categoryKeys) ]); return jsonOut({ ok: true, message: "تم تسجيل الحجز" }); } finally { lock.releaseLock(); } } // ====================================== // التحقق من صلاحية الإدارة // ====================================== if (!checkPin(data.pin)) { return jsonOut({ ok: false, message: "الرقم السري غير صحيح" }); } // ====================================== // حفظ الإعدادات والخدمات والعروض // ====================================== if (action === "save") { if (!data.cfg || typeof data.cfg !== "object") { return jsonOut({ ok: false, message: "بيانات الإعدادات غير صحيحة" }); } saveConfig(ss, data.cfg); return jsonOut({ ok: true, message: "تم حفظ الإعدادات" }); } // ====================================== // حظر موعد لقسم محدد // ====================================== if (action === "block") { var date = String(data.date || ""); var time = String(data.time || ""); var categoryKey = String(data.categoryKey || "*"); if (!date || !time) { return jsonOut({ ok: false, message: "حددي التاريخ والوقت" }); } var lock = LockService.getScriptLock(); lock.waitLock(20000); try { var sh = ss.getSheetByName("Bookings"); var booked = getBookedMap(ss); var day = booked[date] || {}; if ( day[categoryKey] && day[categoryKey].indexOf(time) !== -1 ) { return jsonOut({ ok: false, message: "هذا الموعد محجوز أو محظور بالفعل" }); } sh.appendRow([ Utilities.getUuid(), date, time, "مغلق من الإدارة", "", "وقت محظور", "", "blocked", JSON.stringify([categoryKey]) ]); return jsonOut({ ok: true }); } finally { lock.releaseLock(); } } // ====================================== // فك حظر موعد لقسم محدد // ====================================== if (action === "unblock") { var date = String(data.date || ""); var time = String(data.time || ""); var categoryKey = String(data.categoryKey || "*"); var sh = ss.getSheetByName("Bookings"); var vals = sh.getDataRange().getValues(); for (var i = vals.length - 1; i >= 1; i--) { var rowDate = formatDate(vals[i][1]); var rowTime = formatTime(vals[i][2]); var status = String(vals[i][7] || ""); var keys = parseCategoryKeys(vals[i][8]); if ( rowDate === date && rowTime === time && status === "blocked" && keys.indexOf(categoryKey) !== -1 ) { sh.deleteRow(i + 1); return jsonOut({ ok: true }); } } return jsonOut({ ok: false, message: "لم يتم العثور على الحظر" }); } // ====================================== // حذف حجز // ====================================== if (action === "deleteBooking") { var id = String(data.id || ""); var sh = ss.getSheetByName("Bookings"); var vals = sh.getDataRange().getValues(); for (var i = 1; i < vals.length; i++) { if (String(vals[i][0]) === id) { sh.deleteRow(i + 1); return jsonOut({ ok: true }); } } return jsonOut({ ok: false, message: "الحجز غير موجود" }); } // ====================================== // رفع الصور إلى Google Drive // ====================================== if (action === "upload") { if (!data.data) { return jsonOut({ ok: false, message: "لم يتم اختيار صورة" }); } var bytes = Utilities.base64Decode(data.data); var blob = Utilities.newBlob( bytes, data.mimeType || "image/jpeg", "laverne_" + Date.now() + ".jpg" ); var folderName = "صور مركز لافيرن"; var folders = DriveApp.getFoldersByName(folderName); var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName); var file = folder.createFile(blob); file.setSharing( DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW ); var url = "https://drive.google.com/uc?export=view&id=" + file.getId(); return jsonOut({ ok: true, url: url }); } return jsonOut({ ok: false, message: "الأمر غير معروف" }); } catch (err) { return jsonOut({ ok: false, message: err.toString() }); } } // ========================================== // التحقق من الرقم السري // ========================================== function checkPin(pin) { return String(pin || "") === String(ADMIN_PIN); } // ========================================== // قراءة الإعدادات // ========================================== function getConfig(ss) { var sh = ss.getSheetByName("Config"); if (!sh || sh.getLastRow() < 2) { return null; } var vals = sh.getDataRange().getValues(); for (var i = 1; i < vals.length; i++) { if (String(vals[i][0]) === "cfg") { try { return JSON.parse(vals[i][1]); } catch (e) { return null; } } } return null; } // ========================================== // حفظ الإعدادات دون تغيير بيانات الحجوزات // ========================================== function saveConfig(ss, cfg) { var sh = ss.getSheetByName("Config"); if (!sh) { sh = ss.insertSheet("Config"); } sh.clear(); sh.appendRow(["Key", "Value"]); sh.appendRow(["cfg", JSON.stringify(cfg)]); } // ========================================== // خريطة المواعيد المستقلة لكل قسم // ========================================== function getBookedMap(ss) { var sh = ss.getSheetByName("Bookings"); var vals = sh.getDataRange().getValues(); var map = {}; for (var i = 1; i < vals.length; i++) { var date = formatDate(vals[i][1]); var time = formatTime(vals[i][2]); var status = String(vals[i][7] || "active"); if (!date || !time) continue; var keys = parseCategoryKeys(vals[i][8]); if (!keys.length) { keys = ["*"]; } if (!map[date]) { map[date] = {}; } keys.forEach(function(key) { if (!map[date][key]) { map[date][key] = []; } if (map[date][key].indexOf(time) === -1) { map[date][key].push(time); } }); } return map; } // ========================================== // قراءة مفاتيح الأقسام // ========================================== function parseCategoryKeys(value) { if (Array.isArray(value)) return value; try { var parsed = JSON.parse(String(value || "[]")); if (Array.isArray(parsed)) { return parsed.map(String); } } catch (e) {} return []; } // ========================================== // قائمة الحجوزات للإدارة // ========================================== function getRows(ss) { var sh = ss.getSheetByName("Bookings"); var vals = sh.getDataRange().getValues(); var rows = []; for (var i = 1; i < vals.length; i++) { var date = formatDate(vals[i][1]); var time = formatTime(vals[i][2]); if (!date || !time) continue; rows.push({ id: String(vals[i][0] || ""), d: date, t: time, n: String(vals[i][3] || ""), p: String(vals[i][4] || "").replace(/^'/, ""), svc: String(vals[i][5] || ""), totalText: String(vals[i][6] || ""), status: String(vals[i][7] || "active"), categoryKeys: parseCategoryKeys(vals[i][8]) }); } return rows; } // ========================================== // تنسيق التاريخ // ========================================== function formatDate(v) { if (v instanceof Date) { return Utilities.formatDate( v, Session.getScriptTimeZone(), "yyyy-MM-dd" ); } return String(v || "").trim(); } // ========================================== // تنسيق الوقت // ========================================== function formatTime(v) { if (v instanceof Date) { return Utilities.formatDate( v, Session.getScriptTimeZone(), "HH:mm" ); } return String(v || "").trim(); } // ========================================== // إخراج JSON // ========================================== function jsonOut(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
